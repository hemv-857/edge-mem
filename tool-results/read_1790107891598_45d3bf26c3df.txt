   248→class EdgeConfig:
   249→    """Configuration for creating a new Qdrant Edge shard."""
   250→
   251→    def __init__(
   252→        self,
   253→        vectors: Optional[
   254→            Union["EdgeVectorParams", Dict[str, "EdgeVectorParams"]]
   255→        ] = None,
   256→        sparse_vectors: Optional[Dict[str, "EdgeSparseVectorParams"]] = None,
   257→        on_disk_payload: Optional[bool] = None,
   258→        hnsw_config: Optional["HnswIndexConfig"] = None,
   259→        quantization_config: Optional[QuantizationConfigType] = None,
   260→        optimizers: Optional["EdgeOptimizersConfig"] = None,
   261→        max_search_threads: Optional[int] = None,
   262→        search_pool_core: Optional[int] = None,
   263→    ) -> None:
   264→        """
   265→        Create an EdgeConfig.
   266→
   267→        Parameters left as None are "not specified": when loading an existing shard each
   268→        one resolves through provided -> persisted -> derived from segments -> default,
   269→        so an unspecified parameter keeps the shard as it is. vectors and sparse_vectors
   270→        define the stored data: if provided they are validated for compatibility against
   271→        the existing segments, if omitted they are inherited from the shard.
   272→
   273→        Args:
   274→            vectors: Dense vector configuration. Can be a single EdgeVectorParams for
   275→                     the default vector (name "") or a dict of name -> EdgeVectorParams.
   276→                     Optional if sparse_vectors is provided (sparse-only config).
   277→            sparse_vectors: Optional sparse vector configurations.
   278→            on_disk_payload: If True, store payload on disk (mmap); otherwise in RAM.
   279→                             None keeps the shard's current value (defaults to on-disk).
   280→            hnsw_config: Optional global HNSW config (used when building HNSW index).
   281→            quantization_config: Optional global quantization config.
   282→            optimizers: Optional optimizer settings.
   283→            max_search_threads: Number of threads in the shard's search thread pool, which
   284→                                runs per-segment reads in parallel and loads segments in
   285→                                parallel. None (the default) derives the count from the number
   286→                                of CPUs, matching the core search runtime.
   287→            search_pool_core: Pin every search pool thread to this CPU core (best-effort),
   288→                              bounding search compute to one core. None = OS scheduling.
   289→        """
   290→        ...
   291→
   292→    @property
   293→    def vectors(self) -> Dict[str, "EdgeVectorParams"]:
   294→        """Dense vector configurations."""
   295→        ...

   297→    @property
   298→    def sparse_vectors(self) -> Dict[str, "EdgeSparseVectorParams"]:
